Jevy a kombinatorika

\[ \begin{aligned} p(\bar A) &= 1 - p(A) \\ p(A \cup B) &= p(A) + p(B) - p(A \cap B) \\ p(A \mid B) &= \frac{p(A \cap B)}{p(B)} \\ \text{nezávislé: } p(A \cap B) &= p(A)\,p(B) \end{aligned} \] \[ \binom{n}{k} = \frac{n!}{k!\,(n-k)!}, \qquad V_k(n) = \frac{n!}{(n-k)!}, \qquad V'_k(n) = n^k \] \[ p(A) = \frac{m}{n}, \qquad p(A) = \frac{|A|}{|\Omega|} \ \ \text{(geometrická: délka, plocha)} \]

Úplná pravděpodobnost, Bayes, opakování

\[ \begin{aligned} p(A) &= \sum_i p(H_i)\,p(A \mid H_i), \qquad \textstyle\sum_i p(H_i) = 1 \\ p(H_k \mid A) &= \frac{p(H_k)\,p(A \mid H_k)}{\sum_i p(H_i)\,p(A \mid H_i)} \end{aligned} \] \[ \begin{aligned} p(A \text{ dřív než } B) &= \frac{p(A)}{p(A) + p(B)} \\ p(\text{začínající vyhraje}) &= \frac{a}{1 - (1-a)(1-b)} \end{aligned} \] \[ \sum_{k=0}^{\infty} q^k = \frac{1}{1-q}, \qquad \sum_{k=0}^{\infty} \frac{x^k}{k!} = e^x \]

Náhodná veličina

\[ \begin{aligned} F(x) &= p(X < x), \qquad p(a \le X < b) = F(b) - F(a) \\ p(X = x_i) &= F(x_i^+) - F(x_i) \end{aligned} \] \[ \begin{aligned} EX &= \textstyle\sum x_i p_i \quad\text{nebo}\quad \int x f(x)\,dx \\ EX^2 &= \textstyle\sum x_i^2 p_i \quad\text{nebo}\quad \int x^2 f(x)\,dx \\ DX &= EX^2 - (EX)^2, \qquad \sigma = \sqrt{DX} \end{aligned} \] \[ \begin{aligned} E(aX + b) &= a\,EX + b, \qquad D(aX + b) = a^2 DX \\ E(X + Y) &= EX + EY \\ \text{nezávislé: } D(X + Y) &= DX + DY \end{aligned} \] \[ \begin{aligned} \int_{-\infty}^{\infty} f(x)\,dx &= 1, \qquad f \ge 0, \qquad F(x) = \int_{-\infty}^{x} f(t)\,dt \\ p(a < X < b) &= \int_a^b f(x)\,dx, \qquad F(\tilde x) = 0{,}5 \end{aligned} \]

\[ \begin{aligned} \int x^n\,dx &= \frac{x^{n+1}}{n+1}, \qquad \int e^{ax} dx = \frac{e^{ax}}{a} \\ \int \frac{dx}{x} &= \ln|x| \\ \int u v'\,dx &= uv - \int u' v\,dx \\ \int x \sin x\,dx &= \sin x - x \cos x \\ \int x \cos x\,dx &= \cos x + x \sin x \\ \int x^2 \sin x\,dx &= -x^2 \cos x + 2x \sin x + 2 \cos x \\ \int x^2 \cos x\,dx &= x^2 \sin x + 2x \cos x - 2 \sin x \\ \int x e^{-\lambda x} dx &= -\frac{e^{-\lambda x}}{\lambda^2}(\lambda x + 1) \\ \int x^2 e^{-\lambda x} dx &= -\frac{e^{-\lambda x}}{\lambda^3}(\lambda^2 x^2 + 2\lambda x + 2) \end{aligned} \]

Normální rozdělení

\[ \begin{aligned} u &= \frac{x - \mu}{\sigma}, \qquad p(X < x) = \Phi(u) \\ \Phi(-u) &= 1 - \Phi(u) \\ p(a < X < b) &= \Phi\!\left(\tfrac{b - \mu}{\sigma}\right) - \Phi\!\left(\tfrac{a - \mu}{\sigma}\right) \\ x_p &= \mu + \sigma u_p, \qquad u_{1-p} = -u_p \\ p(|X - \mu| < \sigma u_{(1+\gamma)/2}) &= \gamma \end{aligned} \]

Limitní věty

\[ \begin{aligned} \text{Bi}(n, p) &\approx \text{No}\big(np,\ \sqrt{npq}\big), \qquad npq > 9 \\ \tfrac{X}{n} &\approx \text{No}\big(p,\ \sqrt{pq/n}\big) \\ \textstyle\sum_{i=1}^{n} X_i &\approx \text{No}\big(n\mu,\ \sigma\sqrt{n}\big) \\ \bar X &\approx \text{No}\big(\mu,\ \sigma/\sqrt{n}\big) \\ p(|X - EX| \ge \varepsilon) &\le \frac{DX}{\varepsilon^2} \end{aligned} \]

Náhodný vektor

\[ \begin{aligned} \iint f(x, y)\,dx\,dy &= 1 \\ p((X, Y) \in G) &= \iint_G f(x, y)\,dx\,dy \\ f_X(x) &= \int f(x, y)\,dy, \qquad f_Y(y) = \int f(x, y)\,dx \\ \text{nezávislé} &\iff f(x, y) = f_X(x)\,f_Y(y) \\ \text{cov}(X, Y) &= E(XY) - EX\,EY \\ \rho &= \frac{\text{cov}(X, Y)}{\sigma_X \sigma_Y} \end{aligned} \]

Výběrové charakteristiky

\[ \begin{aligned} \bar x &= \frac{1}{n}\sum x_i \\ s^2 &= \frac{1}{n-1}\sum (x_i - \bar x)^2 = \frac{\sum x_i^2 - (\sum x_i)^2/n}{n-1} \\ \text{četnosti: } \bar x &= \frac{\sum x_j n_j}{n}, \qquad \sigma^2 = \frac{\sum x_j^2 n_j}{n} - \bar x^2 \end{aligned} \]

Intervaly spolehlivosti (1 − α)

\[ \begin{aligned} \mu &\in \bar x \pm t_{1-\alpha/2}(n-1)\,\frac{s}{\sqrt{n}} \\ \mu &\in \bar x \pm u_{1-\alpha/2}\,\frac{\sigma}{\sqrt{n}} \quad (\sigma \text{ známé}) \\ \sigma^2 &\in \left( \frac{(n-1)s^2}{\chi^2_{1-\alpha/2}(n-1)},\ \frac{(n-1)s^2}{\chi^2_{\alpha/2}(n-1)} \right) \end{aligned} \]

Testy (K1–K5)

Oboustranný test: kvantil \(1 - \alpha/2\); jednostranný: \(1 - \alpha\).

\[ T = \frac{\bar x - \mu_0}{s}\sqrt{n} \sim t(n-1) \] \[ U = \frac{\bar x - \mu_0}{\sigma}\sqrt{n} \sim \text{No}(0, 1) \] \[ d_i = x_i - y_i: \quad T = \frac{\bar d}{s_d}\sqrt{n} \sim t(n-1) \] \[ s_p^2 = \frac{(n_1 - 1)s_1^2 + (n_2 - 1)s_2^2}{n_1 + n_2 - 2} \] \[ T = \frac{\bar x_1 - \bar x_2}{s_p\sqrt{\frac{1}{n_1} + \frac{1}{n_2}}} \sim t(n_1 + n_2 - 2) \] \[ U = \frac{\bar x_1 - \bar x_2}{\sqrt{\frac{s_1^2}{n_1} + \frac{s_2^2}{n_2}}} \approx \text{No}(0, 1) \] \[ F = \frac{s_1^2}{s_2^2} \sim F(n_1 - 1,\ n_2 - 1), \qquad s_1^2 \ge s_2^2 \] \[ \chi^2 = \frac{(n-1)s^2}{\sigma_0^2} \sim \chi^2(n-1) \]

Zamítáme \(H_0\): \(|T| \ge t_{1-\alpha/2}\); \(T \le -t_{1-\alpha}\); \(T \ge t_{1-\alpha}\); \(F \ge F_{1-\alpha/2}\).

Testy dobré shody

\[ \begin{aligned} \chi^2 &= \sum_{j=1}^{k} \frac{(n_j - E_j)^2}{E_j}, \qquad E_j = n\,p_j \ge 5 \\ \text{zamítáme: } \chi^2 &\ge \chi^2_{1-\alpha}(k - 1 - r) \end{aligned} \]

r = počet odhadnutých parametrů (Po: \(\lambda = \bar x\), r = 1; No: \(\mu = \bar x\), \(\sigma\), r = 2).

\[ \text{kontingenční tabulka: } E_{ij} = \frac{r_i\,s_j}{n}, \quad \nu = (r - 1)(s - 1) \] \[ \begin{aligned} \text{K–S: } F_n(x_j) &= \frac{n_1 + \dots + n_j}{n} \\ D &= \max_j \big|F_n(x_j) - F_0(x_j)\big| \end{aligned} \] \[ \begin{aligned} &\text{zamítáme: } D \ge D_n(\alpha) \\ &n > 40\text{: } D_n(0{,}05) \approx \frac{1{,}36}{\sqrt{n}}, \quad D_n(0{,}01) \approx \frac{1{,}63}{\sqrt{n}} \end{aligned} \]
